cmake_minimum_required(VERSION 3.15)
project(libzupt_consumer LANGUAGES CXX)
find_package(libzupt CONFIG REQUIRED)
enable_testing()
foreach(kind shared static)
    add_executable(consumer_${kind} main.cpp)
    target_link_libraries(consumer_${kind} PRIVATE libzupt::zupt_${kind})
    target_compile_definitions(consumer_${kind} PRIVATE EXPECTED_VERSION="${libzupt_VERSION}")
    add_test(NAME consumer_${kind} COMMAND consumer_${kind})
endforeach()
